7. Armstrong No

Ex : Print all armstrong numbers from 3 digits, 4 digits and five digits
--------------------------------
 import java.util.Scanner;

class Codechef1 {
    public static void main(String[] args) throws java.lang.Exception {

        for (int no = 100; no < 100000; no++) {
            int sum = 0, m= no;
            String str = String.valueOf(no);

            while (m> 0) {
                int a= m% 10;
                sum += (int) Math.pow(a, str.length());
                m/= 10;
            }

            if (sum == no) {
                System.out.println("Armstrong : " + no);
            }
        }
    }

8. Perfect Number
6 = 1 , 2 , 3  = 6
28 = 1 + 2 + 4 + 7 + 14 = 28

Sum of factor number (exclusive given number ) is same as given number.

import java.util.Scanner;
class Codechef
{
public static void main (String[] args) throws java.lang.Exception
{
Scanner sc=new Scanner(System.in);
int no=sc.nextInt();
int sum=0;
for(int i=1;i<no;i++)
{ 
	if(no%i ==0)
	sum=sum+i;
}
if(sum==no)
System.out.println("Perfect No : " + no); 
else
System.out.println("Not Perfect No : " + no); 
}
}

Triangular Number
Input:3    |     6
output:    |     *
  *        |    * *
 * *       |   * * *

public static void main (String[] args)
	{
	    
		Scanner sc=new Scanner(System.in);
		int n=sc.nextInt();

        int starsprinted=0;
        
        for(int i=0;starsprinted<n;i++){
            for(int j=1;j<=n-i;j++){
                System.out.print(" ");
            }
            for(int j=1;j<=i && starsprinted<=n;j++){
                System.out.print("* ");
                starsprinted++;
            }
            System.out.println();
        }

	}
