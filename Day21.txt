import java.util.*;
import java.lang.*;
import java.io.*;

class A extends Thread{ // inheritng the class 
    public void run(){  // need to use the same signature
        System.out.println("From thread A");
        for(int i=0;i<=10;i++){
            System.out.print("i is: "+i+", ");
        }
        System.out.println("Exit from thread A");
    }
}

class B extends Thread{ // inheritng the class 
    public void run(){  // need to use the same signature
        System.out.println("From thread B");
        for(int j=1;j<=10;j++){
            System.out.print("j is: "+j+", ");
        }
        System.out.println("Exit from thread B");
    }
}                                                      

class C extends Thread{ // inheritng the class 
    public void run(){  // need to use the same signature
        System.out.println("From thread C");
        for(int k=1;k<=10;k++){
            System.out.print("k is: "+k+", ");
        }
        System.out.println("Exit from thread C");
    }
}

class Main{
    public static void main(String args[]){
        A a=new A();
        B b=new B();
        C c=new C();
        
        a.start();
        b.start();
        c.start();
        
    }
}
