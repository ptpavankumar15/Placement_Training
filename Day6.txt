import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{

		Scanner sc=new Scanner(System.in);
		int n=sc.nextInt();
		
		long[][] bell=new long[n+1][n+1];
		bell[0][0]=1;
		
		for(int i=1;i<=n;i++){
		    bell[i][0] = bell[i-1][i-1];
		    for(int j=1;j<=i;j++){
		        bell[i][j]=bell[i-1][j-1]+bell[i][j-1];
		    }
		}
		
		System.out.println(bell[n][0]);

	}
}

public static void main (String[] args) throws java.lang.Exception
	{
		Scanner sc=new Scanner(System.in);
		
        int n=sc.nextInt();
        long terms[]=new long[n];
        terms[0]=2;
        
        System.out.println(terms[0]);
        for(int i=1;i<n;i++){
            long product=1;
            for(int j=0;j<i;j++){
                
                product*=terms[j];
            }
            terms[i]=product+1;
            System.out.print(terms[i]+" ");
        }

	}
