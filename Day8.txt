import java.util.Scanner;
class Main{
public static void main(String args[]){
int arr1[]={1,2,3,4,5};
int arr2[]=arr1;                //assigning - same reference of arr1

int arr3=arr1.clone();          //clone method - different reference --->> copying 

System.out.println("reference of arr1: "+arr1);
for(int i:arr1){
System.out.print(i+" ");
}
System.out.println("reference of arr2: "+arr2);
for(int i:arr2){
System.out.print(i+" ");
}
System.out.println("reference of arr3: "+arr3);
for(int i:arr3){
System.out.print(i+" ");
}

}}
